二、核心数据模型
数据库表结构（SQLite）
sql
-- 概念表：存储所有知识点
CREATE TABLE concepts (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    title TEXT NOT NULL,              -- 概念名，如"机会成本"
    user_definition TEXT,              -- 用户自己的理解（"我终于搞懂了……"）
    source_text TEXT,                  -- 来源原文片段
    mastery TEXT DEFAULT '学习中',    -- 学习中/模糊/搞懂了
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 追问记录表：存储每次追问和回答
CREATE TABLE追问记录 (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    concept_id INTEGER NOT NULL,
    question TEXT NOT NULL,            -- AI问的什么
    user_answer TEXT,                  -- 用户怎么答的
    is_correct BOOLEAN,                -- 方向是否正确
    hint_used BOOLEAN DEFAULT 0,       -- 是否用了提示
    asked_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (concept_id) REFERENCES concepts(id)
);

-- 连接表：知识之间的关系
CREATE TABLE connections (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    concept_a_id INTEGER NOT NULL,
    concept_b_id INTEGER NOT NULL,
    relation_text TEXT NOT NULL,       -- "机会成本关注未来，沉没成本关注过去"
    is_user_edited BOOLEAN DEFAULT 0,  -- 用户是否修改过
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (concept_a_id) REFERENCES concepts(id),
    FOREIGN KEY (concept_b_id) REFERENCES concepts(id),
    UNIQUE(concept_a_id, concept_b_id) -- 防止重复连接
);

-- 每日总结表
CREATE TABLE daily_summaries (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    date DATE DEFAULT CURRENT_DATE,
    concept_id INTEGER,                -- 今天主要学的概念
    breakthrough_text TEXT,            -- "我终于搞懂了……" 
    tomorrow_hook TEXT,                -- 明天AI会追问的问题
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (concept_id) REFERENCES concepts(id)
);

-- 学习设置表（用户配置）
CREATE TABLE settings (
    key TEXT PRIMARY KEY,
    value TEXT,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

# app.py (Streamlit主逻辑)

def main():
    st.title("RecallOS")
    
    # 步骤1：主页 - 输入内容
    if not st.session_state.get("started"):
        source = st.text_area("今天想弄懂什么？")
        if st.button("开始"):
            st.session_state.session = LearningSession(source[:20], source)
            st.session_state.session.start()
            st.session_state.step = "learning"
            st.rerun()
    
    # 步骤2：学习流 - 追问
    elif st.session_state.step == "learning":
        session = st.session_state.session
        
        # 显示当前问题
        question = session.next_question()
        st.markdown(f"🤔 {question}")
        
        # 用户回答
        answer = st.text_input("你的回答：")
        if st.button("发送") or (answer and st.session_state.get("last_input") != answer):
            result = session.submit_answer(answer)
            
            if result["correct"]:
                st.success("✓ 对！")
                st.session_state.depth += 1
            else:
                st.warning(result["feedback"])
                if result["hint"]:
                    st.caption(f"💡 提示：{result['hint']}")
            
            if result["is_done"] or session.depth >= 5:
                st.session_state.step = "connections"
                st.rerun()
    
    # 步骤3：学习流 - 连接
    elif st.session_state.step == "connections":
        connections = session.get_connections()
        for conn in connections:
            st.markdown(f"🔗 {conn['from']} ↔ {conn['to']}")
            st.markdown(f"   {conn['relation']}")
            if st.button("确认这个连接", key=f"confirm_{conn['id']}"):
                save_connection(...)
        
        if st.button("进入总结"):
            st.session_state.step = "summary"
            st.rerun()
    
    # 步骤4：每日总结
    elif st.session_state.step == "summary":
        summary = session.finish()
        st.success(f"✅ 我终于搞懂了……\n\n{summary['breakthrough']}")
        st.caption(f"📌 明天AI会追问你：{summary['tomorrow_hook']}")
        
        if st.button("明天继续"):
            st.session_state.clear()
            st.rerun()

六、开发顺序建议（按优先级）
阶段	任务	时间	产出
第1天	搭建项目结构 + DeepSeek连通	2h	ask_deepseek("你好")能返回
第2天	实现数据库（所有表）	2h	save_concept()和get_all_concepts()能用
第3-4天	实现核心追问逻辑（Prompts）	4-6h	命令行能完成5轮追问
第5天	实现连接推荐逻辑	3-4h	能输出“概念A和概念B有关”
第6-7天	Streamlit界面（5个页面）	4-6h	完整UI，能走通全程
第8天	自己吃狗粮（学3个概念）	2-3h	记录体验问题
第9-10天	调优 + Bug修复	3-4h	可发布版本
总开发时间：25-30小时（跟你之前估计的差不多）
